/** * 處理跨域預檢請求 (CORS Preflight) */ function doOptions(e) { return ContentService.createTextOutput("") .setMimeType(ContentService.MimeType.TEXT) .setHeader("Access-Control-Allow-Origin", "*") .setHeader("Access-Control-Allow-Methods", "POST, OPTIONS") .setHeader("Access-Control-Allow-Headers", "Content-Type"); } /** * 處理正式的 POST 請求 */ function doPost(e) { try { // 1. 解析傳入的 JSON 資料 const params = JSON.parse(e.postData.contents); const name = params.name; const score = params.score; // 取得當下時間 const timestamp = new Date().toLocaleString("zh-TW", { timeZone: "Asia/Taipei" }); // 2. 寫入試算表 const ssId = "YOUR_SPREADSHEET_ID"; // 請在此處填入您的試算表 ID const sheet = SpreadsheetApp.openById(ssId).getActiveSheet(); sheet.appendRow([timestamp, name, score]); // 3. 回傳成功訊息 const result = { status: "success", message: "資料已成功寫入" }; return ContentService.createTextOutput(JSON.stringify(result)) .setMimeType(ContentService.MimeType.JSON) .setHeader("Access-Control-Allow-Origin", "*"); } catch (err) { // 錯誤處理 const errorResult = { status: "error", message: err.toString() }; return ContentService.createTextOutput(JSON.stringify(errorResult)) .setMimeType(ContentService.MimeType.JSON) .setHeader("Access-Control-Allow-Origin", "*"); } }
